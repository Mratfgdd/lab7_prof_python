3. Підключення до БД (database.py) і ORM-моделі

SQLite за замовчуванням не перевіряє FOREIGN KEY, тому прагма вмикається для кожного нового з'єднання через подію connect (експеримент 2). URL береться зі змінної STUDENT_MANAGER_DATABASE_URL або sqlite:///students.db. Сесія створюється з autoflush=False (SQL надсилається явно) і expire_on_commit=False (об'єкти лишаються читабельними після commit).

def make_engine(url: str | None = None, *, echo: bool = False) -> Engine:
    engine = create_engine(url or database_url(), echo=echo)
    if engine.dialect.name == "sqlite":
        event.listen(engine, "connect", _enable_foreign_keys)
    return engine

def _enable_foreign_keys(dbapi_connection: Any, _connection_record: Any) -> None:
    cursor = dbapi_connection.cursor()
    cursor.execute("PRAGMA foreign_keys = ON")
    cursor.close()

def make_session_factory(engine: Engine) -> sessionmaker[Session]:
    return sessionmaker(bind=engine, autoflush=False, expire_on_commit=False)

Моделі написано в стилі SQLAlchemy 2 (Mapped, mapped_column). relationship() з back_populates задає one-to-many; cascade і ondelete="CASCADE" – каскадне видалення. Значення перевіряються двічі: у @validates (до звернення до БД) і CHECK-обмеженнями (навіть якщо запис іде повз ORM).

4. Repository Pattern

Generic Repository містить CRUD, спільний для всіх моделей; конкретні репозиторії додають запити предметної області: пошук за прізвищем, фільтрацію за групою (з JOIN), пошук з фільтрами, сортуванням і пагінацією, AVG, COUNT, MIN/MAX з GROUP BY. Репозиторії викликають лише flush(), а не commit(): межі транзакції визначає сервіс. Фрагмент:

class Repository(Generic[ModelT]):
    model: type[ModelT]

    def add(self, entity: ModelT) -> ModelT:
        self.session.add(entity)
        self.session.flush()
        return entity

    def get_by_id(self, entity_id: int) -> ModelT | None:
        return self.session.get(self.model, entity_id)


class StudentRepository(Repository[Student]):
    model = Student

    def find_by_last_name(self, last_name: str) -> list[Student]:
        statement = (
            select(Student)
            .where(Student.last_name == last_name.strip())
            .order_by(Student.first_name, Student.id)
        )
        return list(self.session.scalars(statement))

    def update_grade(self, student_id: int, grade: float) -> Student | None:
        student = self.get_by_id(student_id)
        if student is None:
            return None
        student.average_grade = grade
        self.session.flush()
        return student


class GroupRepository(Repository[Group]):
    model = Group

    def stats(self) -> list[GroupStats]:
        statement = (
            select(
                Group.name,
                func.count(Student.id),
                func.avg(Student.average_grade),
                func.max(Student.average_grade),
                func.min(Student.average_grade),
            )
            .outerjoin(Student, Student.group_id == Group.id)
            .group_by(Group.id, Group.name)
            .order_by(Group.name)
        )
        return [
            GroupStats(
                name,
                int(students),
                None if average is None else float(average),
                None if best is None else float(best),
                None if worst is None else float(worst),
            )
            for name, students, average, best, worst in self.session.execute(statement)
        ]

7. Транзакція transfer_students, commit і rollback, ACID

transfer_students переводить список студентів до іншої групи як єдине ціле. Після кожного студента виконується flush(), тож UPDATE справді надсилається в БД; якщо наступного студента не знайдено (або виникла помилка БД), rollback() скасовує вже надіслані зміни. Інші операції сервісу виконуються всередині _transaction() – контекстного менеджера з тією самою логікою.

def transfer_students(session: Session, student_ids: Sequence[int], new_group_id: int) -> int:
    unique_ids = list(dict.fromkeys(student_ids))
    try:
        if not unique_ids:
            raise StudentValidationError("список студентів для переведення порожній")
        group = session.get(Group, new_group_id)
        if group is None:
            raise GroupNotFoundError(f"групу з id={new_group_id} не знайдено")
        for student_id in unique_ids:
            student = session.get(Student, student_id)
            if student is None:
                raise StudentNotFoundError(f"студента з id={student_id} не знайдено")
            student.group = group
            session.flush()
        session.commit()
    except Exception:
        session.rollback()
        raise
    return len(unique_ids)

@contextmanager
def _transaction(self) -> Iterator[None]:
    try:
        yield
        self.session.commit()
    except Exception:
        self.session.rollback()
        raise

8. Міграції Alembic

Після alembic init migrations ревізії створено командами alembic revision --autogenerate --rev-id 0001 -m "create groups and students" і ... --rev-id 0002 -m "add student email" (після додавання поля email до моделі). Autogenerate додав колонку й UNIQUE, але не порівнює CHECK-обмеження, тому формат email дописано вручну. На SQLite зміна таблиці виконується через batch («створити нову таблицю → скопіювати дані → замінити»):

def upgrade() -> None:
    with op.batch_alter_table("students", schema=None) as batch_op:
        batch_op.add_column(sa.Column("email", sa.String(length=255), nullable=True))
        batch_op.create_unique_constraint(batch_op.f("uq_students_email"), ["email"])
        batch_op.create_check_constraint(
            op.f("ck_students_email_format"), "email IS NULL OR email LIKE '%_@_%._%'"
        )

def downgrade() -> None:
    with op.batch_alter_table("students", schema=None) as batch_op:
        batch_op.drop_constraint(op.f("ck_students_email_format"), type_="check")
        batch_op.drop_constraint(batch_op.f("uq_students_email"), type_="unique")
        batch_op.drop_column("email")

def test_database_failure_in_the_middle_rolls_back_updates_already_sent(
    school: StudentService,
    session: Session,
    group_of: Callable[[int], str],
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    target = group_id(school, "PI-23")
    real_flush = session.flush
    calls: list[int] = []

    def failing_second_flush(*args: Any, **kwargs: Any) -> None:
        calls.append(1)
        if len(calls) == 2:
            raise OperationalError("UPDATE students", {}, Exception("disk I/O error"))
        real_flush(*args, **kwargs)

    monkeypatch.setattr(session, "flush", failing_second_flush)
    with pytest.raises(OperationalError):
        transfer_students(session, [1, 2, 3], target)
    assert len(calls) == 2
    assert [group_of(1), group_of(2), group_of(3)] == ["PI-21", "PI-21", "PI-22"]

Висновки

Систему обліку студентів (варіант 1) переведено на SQLite: дві пов'язані таблиці з PRIMARY KEY, FOREIGN KEY, UNIQUE і CHECK, SQLAlchemy ORM, Generic Repository, сервіс із транзакціями, пошук через DB-API та дві міграції Alembic. 212 integration тестів на окремій тестовій БД (99.81 % покриття з гілками) перевіряють CRUD, пошук, агрегацію, переведення з rollback, обмеження та міграції. Експерименти показали: FOREIGN KEY у SQLite вмикають явно, rollback захищає від часткових операцій, параметризовані запити відсікають SQL injection, а міграції SQLite потребують транзакційного DDL і перевірки згенерованого коду.
